cmake_minimum_required(VERSION 3.20)
project(Ultima2OriginalWindows LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This bootstrap targets native Win32")
endif()
add_executable(Ultima2Win WIN32 src/main.cpp)
target_compile_features(Ultima2Win PRIVATE cxx_std_17)
target_compile_definitions(Ultima2Win PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(Ultima2Win PRIVATE user32 gdi32 shell32)
if(MSVC)
  target_compile_options(Ultima2Win PRIVATE /W4 /permissive-)
endif()
